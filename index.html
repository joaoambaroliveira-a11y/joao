import random import time import os # Emojis do Tigrão emojis = ["🐅", "🍒", "⭐", "💎", "🔔", "🍋", "7️⃣", "🔥", "🌟", "💰"] def limpar_tela(): os.system('cls' if os.name == 'nt' else 'clear') def jogo_do_tigrao(): print("🐅 BEM-VINDO AO JOGO DO TIGRÃO 🐅") print("=" * 50) banca = 10000.0 # Banca da casa saldo_jogador = 500.0 # Saldo inicial do jogador aposta = 10.0 # Aposta inicial print(f"Banca da casa: R$ {banca:,.2f}") print(f"Seu saldo: R$ {saldo_jogador:,.2f}") print(f"Aposta inicial: R$ {aposta:.2f} (sobe de 10 em 10)\n") while True: limpar_tela() print("🐅 JOGO DO TIGRÃO 🐅") print("=" * 50) print(f"Banca da casa: R$ {banca:,.2f}") print(f"Seu saldo: R$ {saldo_jogador:,.2f}") print(f"Aposta atual: R$ {aposta:.2f}\n") if saldo_jogador < aposta: print("❌ Saldo insuficiente para continuar!") break opcao = input("Pressione ENTER para girar (ou 's' para sair, 'a' para alterar aposta): ").strip().lower() if opcao == 's': break elif opcao == 'a': try: nova_aposta = float(input(f"Digite a nova aposta (múltiplo de 10, mínimo 10): ")) if nova_aposta >= 10 and nova_aposta % 10 == 0 and nova_aposta <= saldo_jogador: aposta = nova_aposta print(f"✅ Aposta alterada para R$ {aposta:.2f}") time.sleep(1.5) continue else: print("❌ Valor inválido!") time.sleep(1.5) continue except: print("❌ Entrada inválida!") time.sleep(1.5) continue # Gira o jogo print("\nGirando os emojis... 🌀") time.sleep(1.5) # Gera 3 emojis aleatórios resultado = random.choices(emojis, k=3) print("\n" + "═" * 30) print(f" {resultado[0]} {resultado[1]} {resultado[2]} ") print("═" * 30 + "\n") # Verifica se tem Tigrão tigrao_count = resultado.count("🐅") # Pagamento aleatório entre 100 e 1000 ganho = random.randint(100, 1000) if tigrao_count >= 1: print(f"🎉 TIGRAÇO APARECEU! Você ganhou R$ {ganho:.2f}!") saldo_jogador += ganho banca -= ganho # Aumenta a aposta automaticamente após vitória if aposta < 200: aposta += 10 print(f"🔥 Aposta subiu para R$ {aposta:.2f}") else: print("😢 Sem Tigrão... Você perdeu a aposta.") saldo_jogador -= aposta banca += aposta # Verifica fim de jogo if banca <= 0: print("🏆 Você quebrou a banca! Parabéns!") break if saldo_jogador <= 0: print("💸 Você ficou sem saldo. Fim de jogo!") break input("\nPressione ENTER para continuar...") print("\n" + "="*50) print(f"Fim de jogo!") print(f"Seu saldo final: R$ {saldo_jogador:,.2f}") print(f"Banca final: R$ {banca:,.2f}") print("🐅 Obrigado por jogar no Tigrão!") if __name__ == "__main__": jogo_do_tigrao()
